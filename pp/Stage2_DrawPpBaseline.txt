nclude "TFile.h"
#include "TH1D.h"
#include "TCanvas.h"
#include "TLegend.h"
#include "TLatex.h"
#include "TStyle.h"
#include "TSystem.h"

void Stage2_DrawPpBaseline(const char* inputFile = "Stage2_pp_Baseline.root",
                           const char* outputPrefix = "Stage2_pp_Baseline")
{
  TFile input(inputFile,"READ");
  if (input.IsZombie()) { Error("Stage2_DrawPpBaseline","Cannot read %s",inputFile); return; }
  const char* species[] = {"pi","ka","pr"};
  const char* labels[] = {"#pi^{+}+#pi^{-}","K^{+}+K^{-}","p+#bar{p}"};
  const char* variables[] = {"Z","Jt"};
  const char* xTitles[] = {"z = |p_{h}|/|p_{jet}|","j_{T} (GeV/c)"};
  const char* yTitles[] = {"(1/N_{jet}) dN/dz","(1/N_{jet}) dN/dj_{T} ((GeV/c)^{-1})"};
  gStyle->SetOptStat(0);
  for (Int_t ib = 0; ib < 2; ++ib) for (Int_t iv = 0; iv < 2; ++iv) {
    TCanvas* canvas = new TCanvas(Form("c_%s_%d",variables[iv],ib),"",1500,480);
    canvas->Divide(3,1);
    for (Int_t is = 0; is < 3; ++is) {
      TH1D* raw = dynamic_cast<TH1D*>(input.Get(Form("hRaw%s_%s_jet%d",variables[iv],species[is],ib)));
      TH1D* ue = dynamic_cast<TH1D*>(input.Get(Form("hUe%s_%s_jet%d",variables[iv],species[is],ib)));
      TH1D* corr = dynamic_cast<TH1D*>(input.Get(Form("Dcorr%s_%s_jet%d",variables[iv],species[is],ib)));
      canvas->cd(is+1);
      if (!raw || !ue || !corr) { TLatex().DrawLatexNDC(0.15,0.5,"Histogram missing"); continue; }
      raw->SetLineColor(kBlack); raw->SetMarkerColor(kBlack); raw->SetMarkerStyle(20);
      ue->SetLineColor(kBlue+1); ue->SetMarkerColor(kBlue+1); ue->SetMarkerStyle(24);
      corr->SetLineColor(kRed+1); corr->SetMarkerColor(kRed+1); corr->SetMarkerStyle(21);
      raw->SetTitle(Form("%s, %.0f #leq p_{T}^{jet} < %.0f GeV/c",labels[is],ib ? 6.0 : 4.0,ib ? 9.0 : 6.0));
      raw->GetXaxis()->SetTitle(xTitles[iv]); raw->GetYaxis()->SetTitle(yTitles[iv]);
      raw->Draw("E1"); ue->Draw("E1 SAME"); corr->Draw("E1 SAME");
      TLegend* legend = new TLegend(0.48,0.68,0.88,0.88); legend->SetBorderSize(0);
      legend->AddEntry(raw,"inside jet","lep"); legend->AddEntry(ue,"off-axis UE","lep"); legend->AddEntry(corr,"UE corrected","lep"); legend->Draw();
    }
    canvas->SaveAs(Form("%s_%s_jet%d.png",outputPrefix,variables[iv],ib));
    canvas->SaveAs(Form("%s_%s_jet%d.pdf",outputPrefix,variables[iv],ib));
  }
}
